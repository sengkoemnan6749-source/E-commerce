    USERS {
        UUID id PK "Primary Key"
        VARCHAR full_name "Customer Name"
        VARCHAR email UK "Unique Login"
        VARCHAR password_hash "Encrypted Password"
        VARCHAR phone "Contact Number"
        VARCHAR role "customer | admin | manager"
        TEXT avatar_url "Profile Image"
        TIMESTAMP created_at "Created Time"
    }

    USER_ADDRESSES {
        UUID id PK "Primary Key"
        UUID user_id FK "References USERS(id)"
        VARCHAR label "Home / Studio / Work"
        VARCHAR recipient_name "Receiver Name"
        VARCHAR phone "Contact Phone"
        TEXT address_line "Street Address"
        VARCHAR city "Khan / City"
        VARCHAR province "Phnom Penh"
        VARCHAR postal_code "Postal Code"
        BOOLEAN is_default "Default Flag"
    }

    CATEGORIES {
        VARCHAR id PK "Category ID (audio, keyboards...)"
        VARCHAR name "Display Name"
        VARCHAR slug UK "URL Friendly Slug"
        TEXT description "Category Details"
        TEXT image_url "Header Banner"
        VARCHAR badge "Popular / Hot"
        INT sort_order "Display Priority"
    }

    PRODUCTS {
        VARCHAR id PK "Product Code (prod-001)"
        VARCHAR category_id FK "References CATEGORIES(id)"
        VARCHAR name "Hardware Model Name"
        VARCHAR slug UK "Unique SEO Slug"
        NUMERIC price "Selling Price (USD)"
        NUMERIC original_price "MSRP Comparison"
        INT stock "Available Inventory"
        NUMERIC rating "Average Score (1-5)"
        INT reviews_count "Total Reviews"
        BOOLEAN is_featured "Frontpage Featured"
        JSONB specs "Technical Specifications"
        JSONB variants "Colors & Switches"
    }

    ORDERS {
        VARCHAR id PK "Order ID (ORD-20261022-0001)"
        UUID user_id FK "References USERS(id)"
        VARCHAR customer_name "Recipient Full Name"
        VARCHAR customer_phone "Notification Phone"
        JSONB shipping_address "Full Address Payload"
        VARCHAR payment_method "khqr | card | cod"
        VARCHAR payment_status "pending | paid | failed"
        VARCHAR order_status "confirmed -> delivered"
        NUMERIC subtotal "Items Subtotal"
        NUMERIC shipping_fee "Courier Fee"
        NUMERIC total "Final Total Charge"
        VARCHAR tracking_number "Delivery Tracking"
        VARCHAR carrier_name "GrabExpress / J&T"
    }

    ORDER_ITEMS {
        UUID id PK "Junction Row ID"
        VARCHAR order_id FK "References ORDERS(id)"
        VARCHAR product_id FK "References PRODUCTS(id)"
        VARCHAR product_name "Snapshot Name"
        VARCHAR variant_name "Selected Color/Switch"
        NUMERIC unit_price "Price at Purchase"
        INT quantity "Item Quantity"
        NUMERIC total_price "Line Subtotal"
    }

    PRODUCT_REVIEWS {
        UUID id PK "Review ID"
        VARCHAR product_id FK "References PRODUCTS(id)"
        UUID user_id FK "References USERS(id)"
        VARCHAR user_name "Customer Nickname"
        INT rating "Score (1 to 5 Stars)"
        VARCHAR title "Review Headline"
        TEXT comment "Feedback Content"
        BOOLEAN verified_purchase "Verified Buyer Flag"
    }

    PROMO_CODES {
        VARCHAR code PK "Discount Code (e.g. WELCOME10)"
        NUMERIC discount_amount "Discount Dollar Value"
        NUMERIC minimum_spend "Minimum Cart Value"
        BOOLEAN is_active "Active Status"
        TIMESTAMP expires_at "Expiration Date"
    }
PK: Primary Key (Unique Table Identifier)
FK: Foreign Key (Referential Connection)
1 : N: One-to-Many Relationship
N : N: Many-to-Many Relationship (via ORDER_ITEMS)
