Database Schema Architecture (Tables, Primary Keys, Foreign Keys & Cardinality)

    USERS ||--o{ USER_ADDRESSES : "1 : N (places)"
    USERS ||--o{ ORDERS : "1 : N (places)"
    USERS ||--o{ PRODUCT_REVIEWS : "1 : N (writes)"
    
    CATEGORIES ||--|{ PRODUCTS : "1 : N (contains)"
    
    ORDERS ||--|{ ORDER_ITEMS : "1 : N (has)"
    PRODUCTS ||--o{ ORDER_ITEMS : "1 : N (part of)"
    
    PRODUCTS ||--o{ PRODUCT_REVIEWS : "1 : N (has)"

    USERS {
        UUID id PK "Primary Key"
        VARCHAR full_name "Customer Name"
        VARCHAR email UK "Unique Login"
        VARCHAR password_hash "Encrypted Password"
        VARCHAR phone "Contact Number"
        VARCHAR role "customer / admin"
        TEXT avatar_url "Profile Image"
        TIMESTAMP created_at "Created Time"
    }

    USER_ADDRESSES {
        UUID id PK "Primary Key"
        UUID user_id FK "References USERS(id)"
        VARCHAR label "Home / Work"
        VARCHAR recipient_name "Receiver Name"
        VARCHAR phone "Contact Phone"
        TEXT address_line "Street Address"
        VARCHAR city "Khan / City"
        VARCHAR province "Province"
        VARCHAR postal_code "Postal Code"
        BOOLEAN is_default "Default Flag"
    }

    CATEGORIES {
        VARCHAR id PK "Category ID"
        VARCHAR name "Display Name"
        VARCHAR slug UK "URL Friendly Slug"
        TEXT description "Category Details"
        TEXT image_url "Banner Image"
        VARCHAR badge "Popular / Hot"
        INT sort_order "Sort Order"
    }

    PRODUCTS {
        VARCHAR id PK "Product Code"
        VARCHAR category_id FK "References CATEGORIES(id)"
        VARCHAR name "Product Name"
        VARCHAR slug UK "Unique Slug"
        NUMERIC price "Price ($)"
        NUMERIC original_price "MSRP"
        INT stock "Inventory"
        NUMERIC rating "Rating (1-5)"
        INT reviews_count "Reviews Count"
        BOOLEAN is_featured "Featured"
        JSONB specs "Specifications"
        JSONB variants "Color/Options"
    }

    ORDERS {
        VARCHAR id PK "Order ID"
        UUID user_id FK "References USERS(id)"
        VARCHAR customer_name "Recipient Name"
        VARCHAR customer_phone "Phone"
        JSONB shipping_address "Shipping Address"
        VARCHAR payment_method "khqr / card / cod"
        VARCHAR payment_status "paid / pending"
        VARCHAR order_status "confirmed / shipped"
        NUMERIC subtotal "Subtotal"
        NUMERIC shipping_fee "Shipping"
        NUMERIC total "Total Charge"
        VARCHAR tracking_number "Tracking #"
        VARCHAR carrier_name "Courier"
    }

    ORDER_ITEMS {
        UUID id PK "Item Row ID"
        VARCHAR order_id FK "References ORDERS(id)"
        VARCHAR product_id FK "References PRODUCTS(id)"
        VARCHAR product_name "Product Name"
        VARCHAR variant_name "Variant"
        NUMERIC unit_price "Unit Price"
        INT quantity "Quantity"
        NUMERIC total_price "Line Total"
    }

    PRODUCT_REVIEWS {
        UUID id PK "Review ID"
        VARCHAR product_id FK "References PRODUCTS(id)"
        UUID user_id FK "References USERS(id)"
        VARCHAR user_name "Customer Name"
        INT rating "Rating (1-5)"
        VARCHAR title "Review Title"
        TEXT comment "Feedback Text"
        BOOLEAN verified_purchase "Verified"
    }

    PROMO_CODES {
        VARCHAR code PK "Promo Code"
        NUMERIC discount_amount "Discount ($)"
        NUMERIC minimum_spend "Min Spend ($)"
        BOOLEAN is_active "Active Status"
        TIMESTAMP expires_at "Expires At"
    }
PK: Primary Key
FK: Foreign Key
1 : N: One-to-Many Relationship
N : N: Many-to-Many Relationship (via ORDER_ITEMS)
